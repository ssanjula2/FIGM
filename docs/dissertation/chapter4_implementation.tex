\section{Static Rule Execution Engine}
The FIGM static analysis engine matches extracted Abstract Language Primitives (ALPs) against formal boolean predicates over a normalized node representation. As of the AST migration documented in \texttt{docs/architecture/AST\_MIGRATION\_MAP.md}, this normalized representation is populated by two distinct mechanisms depending on rule and framework: 15 of the 25 FIGM rules are detected via genuine TypeScript-compiler-API AST traversal for React, Next.js, plain JS/TS, and the TypeScript class body of Angular components (the frameworks/syntax for which a real parse is possible without adding a new dependency); the remaining rules, and every rule applied to Vue SFC markup or Angular's inline template markup (neither of which the TypeScript compiler can parse), are detected via a regex/substring text-heuristic over the source. This is not a uniform "unified AST traversal" claim -- it is a framework- and rule-gated hybrid, and every emitted finding is labeled with which of the two mechanisms produced it (\texttt{sourceKind}: \texttt{REAL\_AST} or \texttt{LEGACY\_TEXT\_HEURISTIC}) so this distinction is verifiable in the tool's own output, not only in this text.

\subsection{Formal Predicate for FIGM-PERF-01}
\begin{equation}
P_PERF_01(n) \iff (type(n) \in {ObjectExpression, ArrayExpression, ArrowFunctionExpression}) \land \exists a \in Ancestors(n): (Scope(a) \in {LOOP, RENDER}) \land \neg UsesLoopVar(n, a)
\end{equation}

\subsection{Formal Predicate for FIGM-PERF-02}
\begin{equation}
P_PERF_02(n) \iff IsVisualElement(n) \land \neg (HasAttr(n, "width") \land HasAttr(n, "height")) \land \neg HasStyle(n, "aspectRatio")
\end{equation}

\subsection{Formal Predicate for FIGM-PERF-03}
\begin{equation}
P_PERF_03(n) \iff IsListenerRegistration(n) \land \neg \exists d \in TeardownScope(Scope(n)): IsPairedRemoval(n, d)
\end{equation}

\subsection{Formal Predicate for FIGM-PERF-04}
\begin{equation}
P_PERF_04(n) \iff (type(n) == ImportDeclaration) \land \exists c \in N: (IsConditionalScope(c) \land ReferencesModule(c, n))
\end{equation}

\subsection{Formal Predicate for FIGM-PERF-05}
\begin{equation}
P_PERF_05(n) \iff IsInlineCallback(n) \land Scope(n) \in {LOOP, RENDER}
\end{equation}

\subsection{Formal Predicate for FIGM-PERF-06}
\begin{equation}
P_PERF_06(n) \iff IsSynchronousStorageAccess(n) \land Scope(n) \in {LOOP, RENDER}
\end{equation}

\subsection{Formal Predicate for FIGM-A11Y-01}
\begin{equation}
P_A11Y_01(n) \iff (IsNonInteractiveElement(n) \land HasPointerHandler(n)) \land \neg (HasKeyboardHandler(n) \land HasTabIndex(n))
\end{equation}

\subsection{Formal Predicate for FIGM-A11Y-02}
\begin{equation}
P_A11Y_02(n) \iff IsCustomControl(n) \land \neg (HasAttr(n, "role") \land (HasAttr(n, "aria-label") \lor HasAttr(n, "aria-labelledby")))
\end{equation}

\subsection{Formal Predicate for FIGM-A11Y-03}
\begin{equation}
P_A11Y_03(n) \iff IsDynamicStatusContainer(n) \land \neg (HasAttr(n, "aria-live") \lor Role(n) \in {"status", "alert"})
\end{equation}

\subsection{Formal Predicate for FIGM-A11Y-04}
\begin{equation}
P_A11Y_04(n) \iff IsOverlayOrModal(n) \land \neg (HasFocusTrap(n) \land RestoresFocusOnUnmount(n))
\end{equation}

\subsection{Formal Predicate for FIGM-A11Y-05}
\begin{equation}
P_A11Y_05(n) \iff IsFormInput(n) \land \neg (HasLabelAssociation(n) \lor HasAriaLabel(n))
\end{equation}

\subsection{Formal Predicate for FIGM-A11Y-06}
\begin{equation}
P_A11Y_06(n) \iff HasTabIndex(n) \land TabIndexValue(n) > 0
\end{equation}

\subsection{Formal Predicate for FIGM-NEXT-01}
\begin{equation}
P_NEXT_01(n) \iff IsReactClientHook(n) \land \neg HasUseClientDirective(File(n))
\end{equation}

\subsection{Formal Predicate for FIGM-NEXT-02}

\subsection{Formal Predicate for FIGM-NEXT-03}

\end{equation}

\subsection{Formal Predicate for FIGM-NEXT-04}

\section{Resilient Source Parsing \& Diagnostics}
The engine handles malformed syntax gracefully, isolating parse errors per component file and returning structured diagnostic reports annotated with WCAG 2.2 and ISO 25010 standard references.
